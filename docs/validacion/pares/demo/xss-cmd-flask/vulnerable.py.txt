import subprocess
from flask import Flask, request

app = Flask(__name__)

@app.route("/hola")
def hola():
    nombre = request.args.get("nombre", "")
    return "<h1>Hola " + nombre + "</h1>"

@app.route("/ping")
def ping():
    host = request.args.get("host")
    salida = subprocess.check_output("ping -c 1 " + host, shell=True)
    return salida
