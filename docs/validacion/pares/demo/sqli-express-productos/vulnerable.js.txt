const express = require("express");
const mysql = require("mysql2");
const db = mysql.createConnection({ host: "localhost", user: "app", database: "tienda" });
const app = express();

app.get("/productos", (req, res) => {
  const categoria = req.query.categoria;
  const orden = req.query.orden || "nombre";
  const sql = "SELECT * FROM productos WHERE categoria = '" + categoria + "' ORDER BY " + orden;
  db.query(sql, (err, filas) => {
    if (err) return res.status(500).send("error");
    res.json(filas);
  });
});

app.listen(3000);
