import org.springframework.web.bind.annotation.*;
import java.sql.*;
import java.util.*;

@RestController
public class UsuarioController {
    private final Connection conn;

    public UsuarioController(Connection conn) { this.conn = conn; }

    @GetMapping("/usuarios")
    public List<String> buscar(@RequestParam String apellido) throws SQLException {
        PreparedStatement ps = conn.prepareStatement("SELECT nombre FROM usuarios WHERE apellido = ?");
        ps.setString(1, apellido);
        ResultSet rs = ps.executeQuery();
        List<String> nombres = new ArrayList<>();
        while (rs.next()) nombres.add(rs.getString(1));
        return nombres;
    }
}
