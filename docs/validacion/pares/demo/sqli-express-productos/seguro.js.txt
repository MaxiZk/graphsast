const express = require("express");
const mysql = require("mysql2");
const db = mysql.createConnection({ host: "localhost", user: "app", database: "tienda" });
const app = express();
const COLUMNAS = ["nombre", "precio", "fecha"];

app.get("/productos", (req, res) => {
  const categoria = req.query.categoria;
  const orden = COLUMNAS.includes(req.query.orden) ? req.query.orden : "nombre";
  db.query("SELECT * FROM productos WHERE categoria = ? ORDER BY " + orden, [categoria], (err, filas) => {
    if (err) return res.status(500).send("error");
    res.json(filas);
  });
});

app.listen(3000);
